CREATE DATABASE IF NOT EXISTS taller_mecanico;
USE taller_mecanico;

CREATE TABLE Usuarios (id INT AUTO_INCREMENT PRIMARY KEY, Nombre VARCHAR(50), Ap VARCHAR(50), Am VARCHAR(50), Telefono VARCHAR(20), Direccion VARCHAR(100));
CREATE TABLE Clientes (id INT AUTO_INCREMENT PRIMARY KEY, Nombre VARCHAR(50), Ap VARCHAR(50), Am VARCHAR(50));
CREATE TABLE Vehiculos (id INT AUTO_INCREMENT PRIMARY KEY, Matricula VARCHAR(20), Modelo VARCHAR(50), Marca VARCHAR(50));
CREATE TABLE Reparaciones (id INT AUTO_INCREMENT PRIMARY KEY, FechaEntrada VARCHAR(20), FechaSalida VARCHAR(20), Falla TEXT, numero_piezas INT);
CREATE TABLE Piezas (id INT AUTO_INCREMENT PRIMARY KEY, Descripcion TEXT, Stock INT);
